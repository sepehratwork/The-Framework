\item \model{Kimi K3}

\paragraph*{مقدمه و ساختار کلان دادگان پس‌آموزش (\en{Post-Training Paradigm})}
فاز پس‌آموزش در مدل زبانی چندوجهی \model{Kimi K3} \cite{kimi2026k3} (با ۲.۸ تریلیون پارامتر کل و ۱۰۴ میلیارد پارامتر فعال در هر توکن) بر پایه یک پارادایم نوین برای مقیاس‌پذیری محاسبات زمان آزمون (\en{Test-Time Scaling}) در پنجره بافت فوق‌طولانی \textbf{یک میلیون توکن} (\en{1M-token context}) طراحی شده است. این فرآیند بر یک زنجیره سه‌مرحله‌ای یکپارچه استوار است:
\begin{itemize}
    \item \textbf{تنظیم دقیق تحت نظارت (\en{SFT}):} ایجاد یک سیاست راه‌انداز سرد (\en{Cold-Start Policy}) با کیفیت فوق‌العاده بالا جهت تثبیت رفتارهای تعاملی و عاملی.
    \item \textbf{یادگیری تقویتی چندسطحی (\en{Multi-Effort RL}):} آموزش متخصصان حوزه‌ای در ۳ دامنه کلان و ۳ سطح تلاش استدلالی (\en{Reasoning Effort}) شامل $\{\text{\en{low}}, \text{\en{high}}, \text{\en{max}}\}$ که به پیدایش ۹ مدل متخصص مجزا منجر می‌شود.
    \item \textbf{تقطیر هم‌سیاست چندمعلمه (\en{Multi-Teacher On-Policy Distillation - MOPD}):} تجمیع و ادغام قابلیت‌های ۹ متخصص حوزه‌ای درون یک مدل واحد سراسری.
\end{itemize}

\paragraph*{سریال‌سازی و ساختار بازنمایی داده‌ها: قالب گفتگوی \en{XTML}}
جهت استانداردسازی تعاملات پیچیده عاملی، داده‌ها با قالب نشانه‌گذاری اختصاصی \en{XTML} (\en{eXtensible Token Markup Language}) فرمت‌بندی شده‌اند. این ساختار با جایگزینی نشانه‌های متداول \en{XML} با نشانه‌های رزرو شده ساختاری شامل \texttt{[open]}، \texttt{[sep]}، \texttt{[close]} و \texttt{[end\_of\_msg]}، ابهام ناشی از توکنایزیشن در مرز المان‌ها را از بین می‌برد و بار مالیات هم‌ترازی (\en{Alignment Tax}) را به حداقل می‌رساند:
\begin{itemize}
    \item \textbf{تفکیک زون‌های داده (\en{Zoning \& Caching}):} پیام‌های گزینه‌ای سراسری (\en{Global Options}) نظیر اعلان ابزارها (\texttt{tool-declare}) و تعیین بودجه تلاش استدلالی (\texttt{thinking-effort}) پیش از تمامی پیام‌های ورودی قرار می‌گیرند تا ساختار حافظه نهان کلید-مقدار (\en{KV Cache}) ابطال نشود. در مقابل، گزینه‌های گذرا (\en{One-Shot}) مانند \texttt{tool\_choice} و \texttt{response\_format} در انتهای پیام‌ها الصاق می‌شوند.
    \item \textbf{معماری سه‌کاناله دستیار (\en{Channels}):} پاسخ مدل دستیار در سه کانال مجزا سازمان‌دهی می‌شود:
    \begin{enumerate}
        \item کانال استدلال و تفکر خام (\texttt{[open]think[sep]}).
        \item کانال پاسخ نهایی کاربر (\texttt{[open]response[sep]}).
        \item کانال فراخوانی ساختاریافته ابزارها (\texttt{[open]tools[sep]}).
    \end{enumerate}
    در این قالب، سازوکار \textbf{تفکر حفظ‌شده} (\en{Preserved Thinking}) اعمال می‌گردد؛ به‌طوری‌که حتی در صورت خالی بودن کانال استدلال، ساختار آن در تاریخچه حفظ می‌شود.
    \item \textbf{فراخوانی ابزار با آرگومان‌های تایپ‌شده (\en{Typed Arguments}):} هر فراخوانی حامل صفت \texttt{tool} و شناسه \texttt{index} برای تطبیق دقیق نتایج فراخوانی‌های موازی است. آرگومان‌های متنی و کدها به صورت متن خام تزریق شده و از ساختارهای پیچیده و فرارداده‌شده \en{JSON} اجتناب می‌گردد. همچنین ماسک‌گذاری زیان (\en{Loss Masking}) بر روی ورودی‌های پشتیبان \en{JSON} اعمال می‌شود.
\end{itemize}

\paragraph*{دادگان پیش‌تنظیم تحت نظارت (\en{Supervised Fine-Tuning - SFT})}
مجموعه‌داده \en{SFT} به منظور ایجاد پایه رفتاری قدرتمند برای فاز یادگیری تقویتی گردآوری شده است:
\begin{itemize}
    \item \textbf{سنتز مسیرهای عاملی (\en{Trajectory Synthesis}):} داده‌های عاملی با استفاده از مدل‌های تخصصی حوزه‌ای نسل‌های پیشین \model{Kimi K2} \cite{kimi2025k2} و \model{Kimi K2.5} \cite{kimi2026k25} شبیه‌سازی و تولید شده‌اند.
    \item \textbf{پالایش و اعتبارسنجی چندمرحله‌ای:} تمامی خط سیرهای تولیدشده از فیلترهای چندمرحله‌ای صحت‌سنجی خودکار و حاشیه‌نویسی انسانی در حلقه (\en{Human-in-the-loop}) عبور داده شدند تا هرگونه انحراف و توهم رفتاری پالایش گردد.
    \item \textbf{انطباق داده‌ها با آموزش آگاه از کوانتایزیشن (\en{QAT}):} داده‌های ورودی همگام با استراتژی کوانتایزیشن مدل در فاز استقرار تنظیم شده‌اند؛ وزن‌های متخصصان \en{MoE} در فرمت \en{MXFP4} و فعال‌سازها در فرمت \en{MXFP8} \cite{rouhani2023mxfp} شبیه‌سازی می‌شوند تا مدل در برابر خطای دقت پایین مقاوم گردد.
\end{itemize}

\paragraph*{سنتز داده‌های یادگیری تقویتی با هدایت گراف دانش (\en{KG-Guided Task Synthesis})}
جهت پوشش جامع حوزه‌های دانشی با عمق مفهومی بالا و اجتناب از داده‌های سطحی، از یک \textbf{گراف جهت‌دار بدون دور (\en{DAG})} خودتوسعه‌دهنده استفاده شده است:
\begin{itemize}
    \item \textbf{کاوش بازگشتی عامل‌ها (\en{Agentic Graph Construction}):} فرآیند با گره‌های هسته در حوزه‌های اصلی (شیمی، فیزیک، ریاضی، علوم کامپیوتر، برنامه‌نویسی و علوم زیستی) آغاز می‌شود. عامل‌ها با وب‌گردی مستقل مفاهیم مرتبط را کشف کرده و یال‌های سلسله‌مراتبی از مفاهیم عام به اتمیک رسم می‌کنند تا یک گراف عمیق شکل گیرد.
    \item \textbf{بازیابی چندسطحی و سنتز وظایف:} با نمونه‌برداری از گره‌های گراف در سطوح دانه‌بندی گوناگون و ادغام با مفاهیم والد، کوئری‌های پیچیده‌ای استخراج شده و با بازیابی مقالات علمی، مخازن کد و اسناد واقعی وب، وظایف استدلالی چالش‌برانگیز تولید می‌شوند.
\end{itemize}

\paragraph*{محیط جعبه‌سفید یکپارچه و تنوع چارچوب‌ها (\en{Unified White-Box RL Environment})}
برای مهار بیش‌برازش مدل به یک قالب یا سیستم پرامپت ثابت، یک محیط چندماژوله جعبه‌سفید طراحی شده که می‌تواند رفتار چارچوب‌های نام‌آشنایی نظیر \en{Kimi Code} \cite{kimicli2026}، \en{Claude Code} \cite{claudecode2026}، \en{Codex} \cite{codex2026}، \en{OpenClaw} \cite{openclaw2026} و \en{Hermes} \cite{hermes2026} را شبیه‌سازی کند. ترکیب پویای ابزارها، مهارت‌ها و حافظه‌ها، استواری مدل را در مواجهه با توزیع‌های مختلف تضمین می‌نماید.

\paragraph*{حوزه‌های تخصصی وظایف و داده‌های یادگیری تقویتی}
داده‌های تقویتی در بسترهای تعاملی و ابزارمحور با ویژگی‌های زیر جمع‌آوری و اعمال شده‌اند:
\begin{itemize}
    \item \textbf{بهینه‌سازی کرنل‌های محاسباتی \en{GPU} (\en{Kernel Tasks}):} داده‌ها از مخازن سطح‌بالای گیت‌هاب نظیر \en{Flash Linear Attention} \cite{yang2024fla} استخراج شده و شامل پیاده‌سازی‌های \en{CUDA}، \en{Triton} \cite{tillet2019triton}، \en{CuTe DSL}، \en{ThunderKittens} \cite{spector2025thunderkittens} و \en{TileLang} \cite{wang2025tilelang} در فرمت‌های \en{BF16}، \en{FP8} و \en{FP4} هستند. تابع پاداش بر مبنای تطابق دقیق خروجی با نسخه مرجع \en{PyTorch} (پاداش صفر در صورت خطای عددی) و نزدیکی به سقف مدل سقف‌شیشه‌ای سخت‌افزار (\en{Hardware Roofline}) بین $0.5$ تا $1.0$ تنظیم شده و سیستم‌های ضدهک پاداش، ترفندهایی مانند \en{CUDA Graph Replay} را جریمه می‌کنند.
    \item \textbf{دستیار شخصی در محیط‌های پایدار (\en{Personal Assistant Workflows}):} با پیاده‌سازی ماک‌های کامل نرم‌افزارهای \en{Gmail}، \en{Notion}، \en{Slack} و \en{Canvas}، سناریوهای شغلی چندروزه با ده‌ها رویداد درهم‌تنیده شبیه‌سازی شده‌اند که هر مسیر اجرایی می‌تواند شامل \textbf{هزاران فراخوانی ابزار} و \textbf{میلیون‌ها توکن زمینه} باشد.
    \item \textbf{وظایف اجرای خودمختار (\en{Autonomous Execution Tasks - AET}):} مدل‌ها با پارادایم تأیید در حلقه (\en{Verify-in-the-Loop}) هدایت می‌شوند. پاداش صرفاً بر اساس بررسی وضعیت نهایی محیط توسط تأییدگرهای مستقل اعطا می‌شود (نه گزارش متنی خود مدل). نمونه این وظایف، مهندسی معکوس و بازسازی سامانه‌های نرم‌افزاری جعبه‌سیاه از طریق ارسال کوئری به اوراکل است.
    \item \textbf{توسعه وب و صحنه‌های تعاملی (\en{Web Development}):} تولید سایت‌های کامل، انیمیشن‌های \en{3D/WebGL}، رابط‌های کاربردی و کدهای \en{SVG}. پاداش شامل آزمون‌های عملکردی قطعی و قضاوت مدل پاداش در بررسی تعاملی و بصری صفحات رندرشده است.
    \item \textbf{استدلال بصری تأییدپذیر (\en{Verifiable Visual Reasoning}):} مدل به مفسر پایتون در سندباکس متصل است و می‌تواند با نوشتن کد، تصاویر مسائل علمی و نمودارها را برش دهد، بزرگ‌نمایی کند و تبدیل‌های هندسی روی آن اعمال نماید تا با دریافت تصاویر حاصله به عنوان مشاهدات جدید، پاسخ را گام‌به‌گام استخراج کند.
\end{itemize}

\paragraph*{فرمول‌بندی‌های ریاضی بودجه، مهار اطناب و تقطیر داده‌ها}
جهت مدیریت طول توکن‌ها و همگرایی پایدار داده‌ها، معادلات زیر بر فرآیند آموزش حاکم شده‌اند:
\begin{itemize}
    \item \textbf{کنترل بودجه استدلال (\en{Reasoning Effort}):} به هر مسئله $x$ یک بودجه توکن اولیه $b_0(x)$ تخصیص یافته و پاداش در صورت فراتر رفتن از حد آستانه با مقدار $-1$ جریمه می‌شود:
    \begin{equation}
        R_{\text{effort}}(x, y) = \begin{cases} 
            R_{\text{task}}(x, y), & \text{if } T(y) \le \tau \cdot b_0(x) \\ 
            -1, & \text{if } T(y) > \tau \cdot b_0(x) 
        \end{cases}
    \end{equation}
    که در آن $T(y)$ در وظایف استدلالی تعداد توکن‌های تفکر و در وظایف عاملی کل توکن‌های خروجی است و ضریب $\tau$ در طول برنامه درسی کاهش می‌یابد.
    \item \textbf{مهار اطناب در مدل پاداش عاملی (\en{GRM}):} اگر طول متن خروجی کاندیدا از $\sigma \cdot \ell_0$ تجاوز کند (که $\ell_0$ طول پاسخ اولیه مدل پایه است)، کاندیدا به صورت قطعی در داوری می‌بازد:
    \begin{equation}
        \text{Length}(y) > \sigma \cdot \ell_0 \implies \text{باخت قطعی در داوری}
    \end{equation}
    \item \textbf{پاداش تقطیر هم‌سیاست چندمعلمه (\en{MOPD}):} هدایت مدل دانشجو $\pi_\theta$ توسط ۹ مدل معلم $\pi_{\text{teacher}}^{(d, e)}$ با پاداش متراکم توکنی زیر انجام می‌شود:
    \begin{equation}
        r_{\text{opd}}^d(y_t \mid e, x, y_{<t}) = \text{clip}\left(\text{sg}\left(\log \frac{\pi_{\text{teacher}}^{(d, e)}(y_t \mid x, y_{<t})}{\pi_\theta(y_t \mid e, x, y_{<t})}\right), -R_{\text{max}}, R_{\text{max}}\right)
    \end{equation}
    که در آن $\text{sg}(\cdot)$ عملگر توقف گرادیان و $R_{\text{max}}$ کران تثبیت است.
    \item \textbf{مدیریت بیات‌شدگی داده‌ها (\en{Data Staleness}):} در فرآیند اجرای جزئی (\en{Partial Rollout})، به محض اتمام نسبت $\lambda \in (0, 1)$ از کل $N \times K$ مسیر نمونه‌برداری‌شده، داده‌ها وارد بهینه‌سازی سیاست می‌شوند و اثر بیات‌شدگی داده‌های معلق با یک منظم‌سازی موضعی در سطح توکن مهار می‌گردد.
\end{itemize}

\paragraph*{آمار زیرساختی و شواهد تجربی ارزیابی دادگان پس‌آموزش}
مقیاس تولید داده و کارایی آن در محیط‌های عملیاتی به شرح زیر ثبت شده است:
\begin{itemize}
    \item در فرآیند پس‌آموزش مدل، در مجموع \textbf{۵۱,۲۱۹,۷۴۱} سندباکس در گستره \textbf{۱,۵۰۵,۶۷۸} ایمیج سیستمی منحصر‌به‌فرد اجرا شده است.
    \item مقیاس‌دهی محاسبات \en{RL} افزایش هم‌زمان و معنادار طول گام‌های تعاملی دستیار (\en{Avg. Steps}) و نرخ موفقیت (\en{Score \%}) را در هر ۸ حوزه عملکردی اصلی نشان می‌دهد.
    \item در آزمون‌های ارزیابی کیفی کورکورانه توسعه وب (\en{Kimi Webdev Bench}) در برابر \en{Claude Opus 4.8}، مدل \model{Kimi K3} در مجموع به برتری قاطع \textbf{+۳۱.۰\%} (با پیروزی در ۵۸.۶٪ پرامپت‌ها) و در حوزه تخصصی گرافیک \en{3D/WebGL} به برتری \textbf{+۵۹.۱\%} دست یافته است (جدول \ref{tab:kimi_k3_webdev}).
\end{itemize}

\paragraph*{مطالعات موردی و توانمندی‌های تعاملی عمیق (\en{Case Studies})}
شواهد تجربی نشان‌دهنده عمق نفوذ داده‌های آموزشی پس‌آموزش در رفتارهای دنیای واقعی مدل است:
\begin{itemize}
    \item \textbf{بهینه‌سازی بلادرنگ کرنل‌های سخت‌افزاری:} مدل تاخیر اجرای کرنل \en{AttnRes} را از ۲۸۳.۶ میلی‌ثانیه به ۱۱۴.۴ میلی‌ثانیه تقلیل داد (\textbf{بهبود +۵۹.۷\%}) و زمان اجرای \en{KDA} را تا ۷۳.۶\% کاهش بخشید.
    \item \textbf{توسعه مستقل کامپایلر \en{MiniTriton}:} مدل به صورت خوداتکا یک کامپایلر کامل شبیه \en{Triton} به همراه کدهای اسمبلی \en{PTX} و زمان اجرای \en{CUDA} ساخت که در ضرب ماتریسی به ۹۰\% توان تئوری سخت‌افزار (\en{cuBLAS}) رسید.
    \item \textbf{طراحی خودکار چیپ \en{Nano-KPU}:} در یک اجرای ۴۸ ساعته خودمختار، مدل یک مدار مجتمع کامل سخت‌افزاری را در کتابخانه ۴۵ نانومتری طراحی، شبیه‌سازی و اعتبارسنجی کرد که در فرکانس ۱۰۰ مگاهرتز با توان عملیاتی بیش از ۸,۷۰۰ توکن بر ثانیه کار می‌کند.
    \item \textbf{پژوهش‌های کلان علمی و تحلیلی:} تحلیل ۳۹۱ رویداد امواج گرانشی با سازمان‌دهی هم‌زمان بیش از ۲۰ ساب‌عامل موازی (\en{Swarm})، و همچنین پردازش بیش از ۱۱,۰۰۰ صفحه گزارش مالی جهت تدوین تاریخچه ۴۲ ساله تراشه‌های \en{ASIC}.
\end{itemize}
