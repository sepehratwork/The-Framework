\item \model{Qwen3}

\paragraph*{مقدمه و ساختار کلی دادگان پیش‌آموزش}
فاز پیش‌آموزش (\en{Pre-training}) در مدل‌های زبانی خانواده \model{Qwen3} با هدف جهش در کارایی محاسباتی، تعمیق قابلیت‌های استدلال ریاضی و برنامه‌نویسی و ارتقای شگرف پوشش چندزبانه طراحی شده است. در این نسخه، مدل‌ها بر روی مقیاس بی‌سابقه‌ای بالغ بر \textbf{۳۶ تریلیون توکن} (\en{36T tokens}) آموزش دیده‌اند که نمایانگر رشد دو برابری نسبت به نسل پیشین (\en{Qwen2.5}) است:
\begin{equation}
    \rho_{\text{tokens}} = \frac{T_{\text{Qwen3}}}{T_{\text{Qwen2.5}}} = \frac{36 \times 10^{12}}{18 \times 10^{12}} = 2.0
\end{equation}
همچنین گستره زبانی دادگان از ۲۹ زبان در نسخه پیشین به \textbf{۱۱۹ زبان و گویش} بسط یافته است که حاکی از افزایش بیش از چهار برابری در تنوع زبان‌شناختی است:
\begin{equation}
    \rho_{\text{lang}} = \frac{N_{\text{lang}}^{\text{Qwen3}}}{N_{\text{lang}}^{\text{Qwen2.5}}} = \frac{119}{29} \approx 4.103
\end{equation}
پیکره پیش‌آموزش به‌گونه‌ای توازن یافته است که از ۶ مدل متراکم (\en{Dense}) در مقیاس‌های $0.6\text{B}$، $1.7\text{B}$، $4\text{B}$، $8\text{B}$، $14\text{B}$ و $32\text{B}$ و دو مدل مبتنی بر ترکیب متخصصان (\en{MoE}) شامل \en{Qwen3-30B-A3B} (با ۳ میلیارد پارامتر فعال) و مدل پرچم‌دار \en{Qwen3-235B-A22B} (دارای ۲۳۵ میلیارد پارامتر کل و ۲۲ میلیارد پارامتر فعال به ازای هر توکن) پشتیبانی نماید. هدف بهینه‌سازی در این مرحله، کمینه‌سازی خطای پیش‌بینی خودبازگشتی توکن بعدی بر روی توالی‌های متنی است:
\begin{equation}
    \mathcal{L}_{\text{pretrain}}(\theta) = - \sum_{t=1}^{T} \log P_\theta(x_t \mid x_1, x_2, \dots, x_{t-1})
\end{equation}

\paragraph*{گردآوری، استخراج چندوجهی و سنتز هدفمند داده‌ها (\en{Data Sourcing \& Synthesis})}
برای تحقق پیکره ۳۶ تریلیون توکنی با چگالی دانشی بالا، از یک خط لوله سه‌گانه و ترکیبی بهره گرفته شده است:
\begin{itemize}
    \item \textbf{استخراج متن اسنادی بر پایه مدل‌های چندوجهی (\en{Multimodal Document Extraction}):} بخش وسیعی از اسناد علمی، دانشگاهی و کتاب‌ها در قالب پرونده‌های \en{PDF} متراکم محبوس مانده‌اند. برای بازیابی دقیق و ساختاریافته متون، ابتدا مدل بینایی-زبانی \en{Qwen2.5-VL} \cite{bai2025qwen25vl} بازتنظیم شد تا نویسه‌خوانی و درک چیدمان بصری اسناد را با دقت بالا انجام دهد. سپس خروجی حاصل با استفاده از مدل \en{Qwen2.5} \cite{yang2024qwen25} بازبینی و پالایش زبانی شد. این فرایند دومرحله‌ای به آزادسازی تریلیون‌ها توکن متنی باکیفیت و با ساختار صحیح انجامید.
    \item \textbf{سنتز داده‌های تخصصی توسط مدل‌های خبره (\en{Domain-Specific Synthetic Data}):} به منظور تعمیق توانایی‌های استدلالی، تریلیون‌ها توکن سنتزی در قالب کتاب‌های درسی، جفت‌های پرسش‌وپاسخ استدلالی، دستورالعمل‌ها و قطعه‌کدهای برنامه‌نویسی به مجموعه افزوده شد. در این مسیر، داده‌های ریاضی به طور ویژه توسط مدل تخصصی \en{Qwen2.5-Math} \cite{yang2024qwen25math} و کدهای چندزبانه به همراه تحلیل الگوریتمی توسط مدل \en{Qwen2.5-Coder} \cite{hui2024qwen25coder} سنتز گردیدند.
    \item \textbf{توسعه چندزبانه متوازن (\en{Multilingual Expansion}):} پوشش زبانی از ۲۹ به ۱۱۹ زبان گسترش یافت؛ داده‌های زبان‌های دارای منابع متوسط و کم با راهبردهای ترجمه معکوس و اعتبارسنجی کیفی تقویت شدند تا نمایش چندزبانه در لایه‌های بازنمایی مدل متوازن گردد.
\end{itemize}

\paragraph*{حاشیه‌نویسی چندزبانه و بهینه‌سازی توزیع داده در سطح نمونه (\en{Instance-Level Data Mixture})}
مدیریت و مهندسی این حجم از داده نیازمند گزینش کیفی فراتر از فیلترینگ‌های سنتی است:
\begin{itemize}
    \item \textbf{سامانه حاشیه‌نویسی چندبعدی:} یک سیستم خودکار برچسب‌گذاری بر روی بیش از \textbf{۳۰ تریلیون توکن} اعمال شد. این سامانه داده‌ها را بر اساس چهار بعد کلیدی ارزیابی و برچسب‌گذاری می‌کند: ارزش آموزشی (\en{Educational Value})، شاخه علمی (\en{Field})، دامنه تخصصی (\en{Domain}) و انطباق با شاخص‌های ایمنی (\en{Safety}).
    \item \textbf{بهینه‌سازی توزیع در سطح نمونه (\en{Instance-Level Optimization}):} برخلاف رویکردهای پیشین نظیر \en{DoREMI} \cite{xie2023doremi}، \en{DoGE} \cite{fan2023doge} و \en{RegMix} \cite{liu2024regmix} که وزن‌دهی به داده‌ها را در سطح منبع داده یا حوزه کلی بهینه‌سازی می‌کردند، در \model{Qwen3} ترکیب داده‌ها بر پایه برچسب‌های ریزدانه در سطح تک‌تک نمونه‌ها تنظیم شد. این فرایند با اجرای مطالعات ابلیشن گسترده بر روی مدل‌های پروکسی کوچک (\en{Small Proxy Models}) انجام گرفت تا بردار ضرایب ترکیب بهینه $w^*$ حاصل شود:
    \begin{equation}
        w^* = \arg\min_{w} \mathbb{E}_{x \sim \mathcal{D}_{\text{eval}}} \left[ \mathcal{L}_{\text{proxy}}(x; \theta^*(w)) \right]
    \end{equation}
\end{itemize}

\paragraph*{برنامه درسی سه‌مرحله‌ای پیش‌آموزش (\en{Three-Stage Pre-training Curriculum})}
فرآیند پیش‌آموزش در قالب سه فاز زمان‌بندی‌شده و ساختاریافته به اجرا درآمد:
\begin{itemize}
    \item \textbf{فاز ۱: مرحله عمومی (\en{General Stage - S1}):} آموزش بر روی بیش از ۳۰ تریلیون توکن با طول زمینه $L = 4{,}096$ توکن اجرا شد. این مرحله پایه دانش زبانی، استنتاج عمومی جهان و تعمیم‌پذیری متقابل در ۱۱۹ زبان و گویش را شکل داد.
    \item \textbf{فاز ۲: مرحله استدلال (\en{Reasoning Stage - S2}):} مدل‌ها با حدود ۵ تریلیون توکن گلچین‌شده و بسیار غنی با محوریت علوم مهندسی، فناوری، ریاضیات (\en{STEM})، مسائل الگوریتمی و داده‌های سنتزی با طول توالی $4{,}096$ آموزش دیدند. در این فاز، نرخ اضمحلال یادگیری (\en{Learning Rate Decay}) تسریع گردید تا مدل در فضاهای استدلالی همگرا شود.
    \item \textbf{فاز ۳: مرحله زمینه طولانی (\en{Long Context Stage}):} مدل‌ها با صدها میلیارد توکن بلند تا سقف طول توالی $L = 32{,}768$ توکن پیش‌آموزش دیدند. ساختار طول اسناد در این فاز شامل \textbf{۷۵٪ داده‌ها با طول بین ۱۶,۳۸۴ تا ۳۲,۷۶۸ توکن} و \textbf{۲۵٪ داده‌ها با طول بین ۴,۰۹۶ تا ۱۶,۳۸۴ توکن} بوده است. در این فاز بسامد پایه فشرده‌سازی موقعیتی دورانی (\en{RoPE}) \cite{su2024roformer} با تکنیک \en{ABF} \cite{xiong2023effective} از ۱۰,۰۰۰ به ۱,۰۰۰,۰۰۰ افزایش یافت و با سازوکارهای \en{YaRN} \cite{peng2023yarn} و توجه چندتکه‌ای دوگانه (\en{DCA}) \cite{an2024dca} ادغام شد تا پنجره بافت حین استنتاج تا سقف ۱۲۸,۰۰۰ توکن ($128\text{K}$) با حفظ پایایی بسط یابد.
\end{itemize}

\paragraph*{توکنایزیشن و بازنمایی زیرکلمات (\en{Tokenization})}
در مدل‌های \model{Qwen3} از توکنایزر اختصاصی مبتنی بر کدگذاری جفت‌بایت در سطح بایت (\en{BBPE}) \cite{bai2023qwen, brown2020language, wang2020neural, sennrich2016neural} با اندازه واژگان $151{,}669$ استفاده شده است. این اندازه واژگان، ضریب فشرده‌سازی توکن به کلمه را در متون غیرانگلیسی به خصوص خطوط خاورمیانه و آسیایی بهینه ساخته و سربار محاسباتی طول توالی را تعدیل می‌کند.

\paragraph*{ارزیابی‌های آماری و مقایسه‌ای مدل‌های پایه (\en{Base Model Evaluations})}
اثربخشی داده‌های پیش‌آموزش و راهبردهای پالایش آن، بر روی ۱۵ بنچمارک استاندارد در ۴ حوزه وظایف عمومی (\en{MMLU} \cite{hendrycks2021mmlu}, \en{MMLU-Redux} \cite{gema2024are}, \en{MMLU-Pro} \cite{wang2024mmlupro}, \en{BBH} \cite{suzgun2023challenging}, \en{SuperGPQA} \cite{du2025supergpqa})، ریاضیات و علوم پایه (\en{GPQA} \cite{rein2023gpqa}, \en{GSM8K} \cite{cobbe2021training}, \en{MATH} \cite{hendrycks2021math})، کدنویسی (\en{EvalPlus} \cite{liu2023evalplus}, \en{MultiPL-E} \cite{cassano2023multipl}, \en{MBPP} \cite{austin2021program}, \en{CRUX-O} \cite{gu2024cruxeval}) و چندزبانه (\en{MGSM} \cite{shi2023language}, \en{MMMLU} \cite{openai2024mmmlu}, \en{INCLUDE} \cite{romanou2024include}) مورد ارزیابی دقیق قرار گرفت. 

جداول \ref{tab:qwen3_235b_base} تا \ref{tab:qwen3_small_base} عملکرد کمّی مدل‌های پایه را در برابر رقبای هم‌رده و پرچم‌دار نشان می‌دهند. مهم‌ترین یافته‌های آماری به شرح زیر است:
\begin{itemize}
    \item \textbf{برتری مدل پرچم‌دار \en{Qwen3-235B-A22B-Base}:} این مدل با دارا بودن تنها ۲۲ میلیارد پارامتر فعال در هر توکن (تقریباً $60\%$ پارامترهای فعال و $35\%$ پارامترهای کل \en{DeepSeek-V3} \cite{liu2024deepseekv3})، در ۱۴ آزمون از ۱۵ آزمون ارزیابی‌شده رتبه اول را کسب کرده است؛ از جمله در آزمون استدلال پیشرفته \en{MATH} با ثبت امتیاز $71.84$ در مقایسه با $62.62$ مدل \en{DeepSeek-V3}، رشد نسبی $+14.72\%$ را محقق ساخته است.
    \item \textbf{انتقال تراز کارایی در مدل‌های متراکم (\en{Dense Models}):} مدل \en{Qwen3-32B-Base} با کمتر از نیمی از پارامترهای مدل پرچم‌دار پیشین \en{Qwen2.5-72B-Base}، در ۱۰ بنچمارک از ۱۵ بنچمارک بر آن غلبه کرده و در کدنویسی (\en{EvalPlus} با امتیاز $72.05$ در برابر $65.93$) و آزمون \en{MMLU-Pro} (امتیاز $65.54$ در برابر $58.07$) برتری قاطعی به ثبت رسانده است. همچنین مدل‌های سبک‌تر \en{Qwen3-8B}، \en{4B} و \en{1.7B} توانسته‌اند مدل‌های مقیاس‌های ۲ تا ۳ برابری نسل قبل خود (\en{Qwen2.5-14B}، \en{7B} و \en{3B}) را در اکثر حوزه‌ها پشت سر بگذارند.
    \item \textbf{بهره‌وری فوق‌العاده مدل‌های \en{MoE} کوچک:} مدل \en{Qwen3-30B-A3B-Base} با تنها ۳ میلیارد پارامتر فعال، در تمامی شاخص‌ها مدل متراکم \en{Qwen2.5-14B-Base} (با ۱۴ میلیارد پارامتر فعال) را مغلوب کرده و به کارایی نزدیک به مدل متراکم ۳۲ میلیارد پارامتری با بیش از ۱۰ برابر هزینه محاسباتی کمتر دست یافته است.
\end{itemize}
