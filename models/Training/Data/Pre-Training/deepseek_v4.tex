\item \model{DeepSeek-V4}

\paragraph*{مقدمه و ساختار کلی دادگان پیش‌آموزش}
فاز پیش‌آموزش (\en{Pre-training}) در مدل‌های زبانی سری \model{DeepSeek-V4} با هدف شکستن سد محاسباتی در پردازش توالی‌های فوق‌طولانی تا سقف بافت یک میلیون توکن (\en{1M-token context}) بازطراحی شده است. در این چارچوب، دو نسخه پایه با مقیاس‌های محاسباتی متمایز آموزش دیده‌اند:
\begin{itemize}
    \item \textbf{مدل \en{DeepSeek-V4-Flash}:} دارای ۲۸۴ میلیارد پارامتر کل (۱۳ میلیارد پارامتر فعال در هر توکن) که روی مجموعه‌ای بالغ بر \textbf{۳۲ تریلیون توکن} (\en{32T tokens}) پیش‌آموزش دیده است.
    \item \textbf{مدل \en{DeepSeek-V4-Pro}:} دارای ۱.۶ تریلیون پارامتر کل (۴۹ میلیارد پارامتر فعال در هر توکن) که فرایند پیش‌آموزش آن بر روی بیش از \textbf{۳۳ تریلیون توکن} (\en{33T tokens}) صورت پذیرفته است.
\end{itemize}

هدف بهینه‌سازی پایه در طول این فرایند، حداقل‌سازی تلفات خودرگرسیو مدل‌سازی زبان (\en{Autoregressive Language Modeling Loss}) بر روی توالی توکن‌ها است:
\begin{equation}
    \mathcal{L}_{\text{LM}}(\theta) = - \sum_{t=1}^{N} \log P_\theta(x_t \mid x_1, x_2, \dots, x_{t-1})
\end{equation}

\paragraph*{ترکیب‌بندی و پالایش پیکره داده‌ها (\en{Data Construction})}
پیکره متنی پیش‌آموزش بر پایه تجربیات نسخه \en{DeepSeek-V3} \cite{deepseek2024v3} شکل گرفته، اما با هدف ارتقای چگالی دانش، جلوگیری از افت کیفی و پشتیبانی از استدلال عمیق، اصلاحات اساسی در حوزه‌های زیر اعمال شده است:
\begin{itemize}
    \item \textbf{پالایش وب و پیشگیری از فروپاشی مدل (\en{Mitigating Model Collapse}):} به منظور مقابله با پدیده فروپاشی مدل ناشی از انباشت داده‌های بازتولیدشده توسط هوش مصنوعی و متون قالبی در وب باز \cite{zhu2024synthesize}، خط لوله‌های فیلترینگ سخت‌گیرانه‌ای تعبیه شد تا محتواهای ماشینی، تکراری و قالبی به صورت دسته‌ای شناسایی و پالایش شوند.
    \item \textbf{تزریق داده‌های عامل‌محور در آموزش میانی (\en{Agentic Mid-Training Data}):} پیکره ریاضیات و برنامه‌نویسی به عنوان هسته اصلی تفکر مدل حفظ شده است؛ علاوه بر آن، در فاز میانی آموزش (\en{Mid-Training Phase}) داده‌های تعاملی و ابزارمحور (\en{Agentic data})—شامل ردپای خط فرمان، محیط‌های چندمرحله‌ای رفع باگ و بازخورد مفسرها—به مدل تزریق شد تا بازنمایی کد از متن ایستا به کنشگری پویا ارتقا یابد.
    \item \textbf{توسعه داده‌های چندزبانه و دانش حاشیه‌ای (\en{Long-Tail Knowledge}):} برای بازنمایی دقیق فرهنگ‌ها و زبان‌های گوناگون، دامنه دادگان چندزبانه توسعه یافت تا پوشش مدل در بازیابی دانش حاشیه‌ای با فراوانی آماری اندک تضمین گردد.
    \item \textbf{تمرکز راهبردی بر اسناد طولانی دانشگاهی (\en{Long-Document Curation}):} به منظور بارورسازی درک زمینه در مقیاس‌های فراتر از چند صدهزار کلمه، مقالات پژوهشی، گزارش‌های تخصصی و اسناد فنی دارای پیوستگی عمیق معنایی با اولویت بالا گزینش شدند.
\end{itemize}

\paragraph*{پیش‌پردازش، توکنایزیشن و مهندسی نمونه‌ها (\en{Data Pre-Processing})}
فرایند مهندسی داده‌ها جهت بهینه‌سازی ورود توکن‌ها به سازوکار توجه شامل راهکارهای زیر است:
\begin{itemize}
    \item \textbf{واژگان و نشانه‌گذار (\en{Tokenizer}):} دایره واژگان در اندازه $128\text{K}$ توکن تثبیت شده و نشانه‌های ویژه‌ای (\en{Special tokens}) جهت ساختاربندی و مدیریت بافت‌های متنی طولانی افزوده شده است.
    \item \textbf{تکمیل درون‌متنی (\en{Fill-in-the-Middle - FIM}):} جهت تعمیق درک دوطرفه متن، راهبرد \en{FIM} \cite{deepseek2024v3} اعمال گردید که طی آن سند متنی به سه بخش پیشوند ($P$)، میانه ($M$) و پسوند ($S$) تفکیک و به فرم زیر بازآرایی می‌شود:
    \begin{equation}
        \mathcal{D}_{\text{FIM}} = [\text{<PRE>}] \circ P \circ [\text{<SUF>}] \circ S \circ [\text{<MID>}] \circ M
    \end{equation}
    این رویکرد مدل را قادر می‌سازد با شرط‌گذاری هم‌زمان روی ابتدا و انتهای بافت، بخش میانی را پیش‌بینی کند.
    \item \textbf{بسته‌بندی اسناد بدون اتلاف توکن (\en{Document Packing}):} با اقتباس از رهیافت دینگ و همکاران \cite{ding2024fewer}، به‌جای قطع نامتقارن (\en{Truncation}) متون در انتهای پنجره بافت، اسناد برگرفته از منابع گوناگون بر پایه طول دسته‌بندی و در قالب بسته‌های بهینه در کنار یکدیگر قرار گرفتند تا کمترین برش رخ دهد.
    \item \textbf{ماسک توجه در سطح نمونه (\en{Sample-Level Attention Masking}):} برخلاف \en{DeepSeek-V3}، در این نسخه ماسک‌گذاری توجه در سطح سند فعال شد. ماتریس توجه $M$ به‌گونه‌ای پیکربندی می‌شود که از سرایت توجه میان اسناد مستقلی که در یک توالی بسته‌بندی شده‌اند جلوگیری شود:
    \begin{equation}
        M_{ij} = \begin{cases} 
            0, & \text{if } i \ge j \text{ and } \text{DocID}(i) = \text{DocID}(j) \\ 
            -\infty, & \text{otherwise} 
        \end{cases}
    \end{equation}
\end{itemize}

\paragraph*{برنامه درسی طول بافت و پویایی بهینه‌سازی (\en{Context Curriculum \& Optimization})}
فرایند آموزش مدل روی بافت‌های میلیونی به شیوه گام‌به‌گام و مرحله‌ای (\en{Curriculum Learning}) مدیریت شده است:
\begin{itemize}
    \item \textbf{توالی افزایش بافت:} آموزش از طول توالی $4\text{K}$ آغاز شده و به توالی‌های $16\text{K}$، سپس $64\text{K}$ و در نهایت $1\text{M}$ توکن تعمیم یافته است.
    \item \textbf{مقیاس دسته‌ها (\en{Batch Size Scheduling}):} در نسخه \en{Flash} اندازه دسته تا سقف $75.5\text{M}$ توکن و در نسخه \en{Pro} تا مرز $94.4\text{M}$ توکن در هر گام افزایش یافته است.
    \item \textbf{ورود تدریجی توجه تنک:} در طول ۱ تریلیون توکن نخست، مدل با سازوکار توجه متراکم (\en{Dense Attention}) گرم‌سازی شد. سپس در طول بافت $64\text{K}$، معماری توجه تنک فشرده (\en{CSA}) با یک مرحله پایدارسازی کوتاه‌مدت برای شاخص‌گذار \en{Lightning Indexer} فعال گردید و تا پایان فاز پیش‌آموزش ادامه یافت.
    \item \textbf{نرخ یادگیری:} مدل‌ها با بهینه‌ساز \en{Muon} \cite{jordan2024muon, liu2025muon} و تنظیمات نرخ یادگیری خطی اولیه و واپاشی کسینوسی نهایی (از $2.7 \times 10^{-4}$ به $2.7 \times 10^{-5}$ در مدل \en{Flash} و از $2.0 \times 10^{-4}$ به $2.0 \times 10^{-5}$ در مدل \en{Pro}) به پیش‌آموزش رسانده شدند.
\end{itemize}

\paragraph*{مهار ناپایداری‌های ناشی از توزیع داده‌ها}
جهت رفع جهش‌های زیان (\en{Loss Spikes}) ناشی از داده‌های پرت در لایه‌های \en{MoE}، دو تکنیک در خط لوله داده و محاسبات اعمال گردید:
\begin{itemize}
    \item \textbf{مسیریابی پیش‌دستانه (\en{Anticipatory Routing}):} شاخص‌های گیتینگ بر اساس وزن‌های با تأخیر تاریخی $\theta_{t-\Delta t}$ روی داده‌های پیش‌واکشی‌شده محاسبه شده تا چرخه هم‌افزایی مثبت مقادیر فرین شکسته شود.
    \item \textbf{تحدید دامنه \en{SwiGLU}:} ورودی‌های خطی فعال‌ساز \en{SwiGLU} \cite{shazeer2020glu} به بازه $[-10, 10]$ و مؤلفه گیت آن به حداکثر مقدار $10$ تحدید شدند.
\end{itemize}

\paragraph*{ارزیابی‌های آماری مدل‌های پایه (\en{Base Model Evaluations})}
نتایج ارزیابی مدل‌های پایه پیش‌آموزش‌دیده (پیش از هرگونه هم‌ترازسازی یا \en{Post-training}) در جدول زیر گزارش شده است. این نتایج نشان می‌دهد که مدل \en{DeepSeek-V4-Flash-Base} با وجود دارا بودن تنها ۱۳ میلیارد پارامتر فعال، به برکت کیفیت داده‌ها و راهبردهای پیش‌پردازش، مدل پیشین \en{DeepSeek-V3.2-Base} (با ۳۷ میلیارد پارامتر فعال) را در بنچمارک‌های متعددی نظیر \en{C-Eval} ($92.1$ در برابر $90.4$)، \en{MultiLoKo} ($42.2$ در برابر $38.7$) و \en{LongBench-V2} ($44.7$ در برابر $40.2$) پشت سر گذاشته است. همچنین مدل \en{DeepSeek-V4-Pro-Base} در آزمون حقیقت‌سنجی دادگان حاشیه‌ای (\en{FACTS Parametric}) به جهش چشمگیر $62.6$ (رشد نسبی ۱۳۱٪ نسبت به $27.1$) دست یافته است.
